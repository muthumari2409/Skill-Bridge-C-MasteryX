#include <stdio.h>
#include <stdlib.h>

int main()
{
    int n, newn, i;
    int *a, *temp;

    printf("Enter the initial number of elements: ");
    scanf("%d", &n);

    a = (int *)malloc(n * sizeof(int));

    if (a == NULL)
    {
        printf("Memory allocation failed");
        return 1;
    }

    printf("Enter %d elements:\n", n);
    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter the new size of the array: ");
    scanf("%d", &newn);

    if (newn < n || newn <= 0)
    {
        printf("New size must be greater than the initial size");
        free(a);
        return 1;
    }

    temp = (int *)realloc(a, newn * sizeof(int));

    if (temp == NULL)
    {
        printf("Memory reallocation failed");
        free(a);
        return 1;
    }

    a = temp;

    printf("Enter %d new elements:\n", newn - n);
    for (i = n; i < newn; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("All elements in the array are:\n");
    for (i = 0; i < newn; i++)
    {
        printf("%d ", a[i]);
    }

    free(a);

    return 0;
}
