#include <stdio.h>

int main()
{
    int n = 6, i, sum = 0;

    for(i = 1; i < n; i++)
    {
        if(n % i == 0)
        {
            sum = sum + i;
        }
    }

    if(sum == n)
        printf("Perfect Number");
    else
        printf("Not a Perfect Number");

    return 0;
}
