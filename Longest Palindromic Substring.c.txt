#include <stdio.h>
#include <string.h>

int main()
{
    char s[100];
    int i, j, k;
    int max = 0, start = 0;
    int flag;

    printf("Enter string: ");
    scanf("%s", s);

    for(i = 0; i < strlen(s); i++)
    {
        for(j = i; j < strlen(s); j++)
        {
            flag = 1;

            for(k = 0; k < (j - i + 1) / 2; k++)
            {
                if(s[i + k] != s[j - k])
                {
                    flag = 0;
                    break;
                }
            }

            if(flag == 1 && j - i + 1 > max)
            {
                max = j - i + 1;
                start = i;
            }
        }
    }

    printf("Longest Palindrome: ");

    for(i = start; i < start + max; i++)
    {
        printf("%c", s[i]);
    }

    return 0;
}
