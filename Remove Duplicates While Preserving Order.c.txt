#include <stdio.h>

int main()
{
    char s[100];
    int i, j, found;

    printf("Enter string: ");
    scanf("%s", s);

    for(i = 0; s[i] != '\0'; i++)
    {
        found = 0;

        for(j = 0; j < i; j++)
        {
            if(s[i] == s[j])
            {
                found = 1;
                break;
            }
        }

        if(found == 0)
        {
            printf("%c", s[i]);
        }
    }

    return 0;
}
