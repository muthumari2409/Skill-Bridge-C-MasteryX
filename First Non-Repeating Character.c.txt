#include <stdio.h>

int main()
{
    char s[100];
    int count[256] = {0};
    int i, found = 0;

    printf("Enter string: ");
    scanf("%s", s);

    for(i = 0; s[i] != '\0'; i++)
    {
        count[s[i]]++;
    }

    for(i = 0; s[i] != '\0'; i++)
    {
        if(count[s[i]] == 1)
        {
            printf("First Non-Repeating Character: %c", s[i]);
            found = 1;
            break;
        }
    }

    if(found == 0)
        printf("-1");

    return 0;
}
