```c
#include <stdio.h>

int main()
{
    int limit, n, temp, digit, sum, count, i, power;

    printf("Enter the limit: ");
    scanf("%d", &limit);

    for (i = 1; i <= limit; i++)
    {
        n = i;
        temp = i;
        count = 0;
        sum = 0;

        while (temp > 0)
        {
            count++;
            temp = temp / 10;
        }

        temp = i;

        while (temp > 0)
        {
            digit = temp % 10;
            power = 1;

            for (int j = 1; j <= count; j++)
                power = power * digit;

            sum = sum + power;
            temp = temp / 10;
        }

        if (sum == n)
            printf("%d ", n);
    }

    return 0;
}
```
